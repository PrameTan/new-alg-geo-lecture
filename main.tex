	\section{Presheaves and Sheaves}
	
	\subsection*{Presheaves}
	\begin{definition}
		Let $X$ be a topological space. A \inlinedef{presheaf of Abelian groups} $\mathcal{F}$ on $X$ consists of the following data:
		\begin{enumerate}
			\item an assignment of an Abelian group $\mathcal{F}(U)$ to each open subset $U \subseteq X$;
			\item for each inclusion of open subsets $U \subseteq V$, there is a group homomorphism $\res^V_U \colon \mathcal{F}(V) \to \mathcal{F}(U)$ such that
			\begin{enumerate}
				\item $\res^U_U = \id_{\mathcal{F}(U)}$ for all open subset $U$;
				\item if $U \subseteq V \subseteq W$ are open subsets, then $\res^W_U = \res^V_U \circ \res^W_V$.
			\end{enumerate}
		\end{enumerate}
	\end{definition}
	
	An element of $\mathcal{F}(U)$ is called a \inlinedef{section} of $\mathcal{F}$ over $U$. The morphism $\res^V_U$ is called the \inlinedef{restriction map} from $V$ to $U$. If $s \in \mathcal{F}(V)$, the more suggestive notation for $\res^V_U(s)$ is $s|_U$. The condition for restriction maps is then $(s|_V)|_U = s|_U$ for all open subsets $U \subseteq V \subseteq W$ and $s \in \mathcal{F}(W)$. Another notation for $\mathcal{F}(U)$ is $\Gamma(U, \mathcal{F})$ which will be useful when we define sheaf cohomology.
	
	\begin{example}\label{chap3:presheaf-of-smooth-functions}
		An example of presheaves on a topological space is the presheaf of real-valued smooth function on a smooth manifold $M$. If $U \subseteq M$ is open, we define $\mathcal{C}^\infty(U)$ to be the set of real-valued smooth functions on $U$. The restriction maps are simply restriction of functions, which is the motivation for the notation $s|_U$.
	\end{example}
	
	We now turn to the definition of morphisms between presheaves.
	
	\begin{definition}
		Given two presheaves $\mathcal{F}$ and $\mathcal{G}$, a \inlinedef{morphism of presheaves} $f \colon \mathcal{F} \to \mathcal{G}$ is the collection of maps $f_U \colon \mathcal{F}(U) \to \mathcal{G}(U)$ for each open subset $U \subseteq X$ such that whenever $U \subseteq V$ are open subsets, the diagram
		% https://q.uiver.app/#q=WzAsNCxbMCwwLCJcXG1hdGhjYWx7Rn0oVikiXSxbMCwyLCJcXG1hdGhjYWx7Rn0oVSkiXSxbMiwyLCJcXG1hdGhjYWx7R30oVSkiXSxbMiwwLCJcXG1hdGhjYWx7R30oVikiXSxbMCwxLCJcXHJlc15WX1UiLDJdLFsxLDIsImZfVSIsMl0sWzAsMywiZl9WIl0sWzMsMiwiXFxyZXNeVl9VIl1d

			{\mathcal{F}(V)} && {\mathcal{G}(V)} \\
			\\
			{\mathcal{F}(U)} && {\mathcal{G}(U)}
			\arrow["{f_V}", from=1-1, to=1-3]
			\arrow["{\res^V_U}"', from=1-1, to=3-1]

		commutes.
	\end{definition}
	
	We can alternatively give a categorical perspective on presheaves. Let $X$ be a topological space. The open sets of $X$ form a category $(\operatorname{Ouv}(X))$ of open subsets of $X$. We declare that there is exactly one morphism $U \to V$ in $(\operatorname{Ouv}(X))$ if and only if $U \subseteq V$. Put simply, this is the order category of open subsets of $X$ with the ordering given by inclusion of sets. Then a presheaf on $X$ is simply a contravariant functor $\mathcal{F} \colon (\operatorname{Ouv}(X))^{\op} \to (\Abs)$. A morphism of presheaves $f \colon \mathcal{F} \to \mathcal{G}$ is then a natural transformation between the two functors.
	
	By changing the target so that $\mathcal{F} \colon (\operatorname{Ouv}(X))^{\op} \to\mathcal{C}$ maps to another category $\mathcal{C}$, we obtain a presheaf $\mathcal{F}$ on $X$ with value in $\mathcal{C}$, not just Abelian groups. In the following we mostly consider the following types of presheaves:
	\begin{enumerate}
		\item Presheaves of sets 
		$(\operatorname{Ouv}(X))^{\op} \to(\Sets)$.
		\item Presheaves of Abelian groups 
		$(\operatorname{Ouv}(X))^{\op} \to(\Abs)$.
		\item Presheaves of rings $(\operatorname{Ouv}(X))^{\op} \to(\Rings)$.
		\item Given a ring $R$, presheaves of $R$-modules $(\operatorname{Ouv}(X))^{\op} \to(R\operatorname{-Mod})$ and presheaves of $R$-algebras $(\operatorname{Ouv}(X))^{\op} \to(R\operatorname{-Alg})$.		
	\end{enumerate}
	
	\begin{example}[Constant Presheaf]
		Let $A$ be a fixed Abelian group and let $X$ be a topological space. We can define a presheaf $\mathcal{A}$ on $X$ by setting $\mathcal{A}(U) = A$ for all open subsets $U \subseteq X$ and all restriction maps are the identity homomorphism. This is called the \inlinedef{constant presheaf} with value $A$ on $X$.
	\end{example}
	
	\begin{example}[Restriction of Presheaf]
		Let $X$ be a topological space and $\mathcal{F}$ a presheaf on $X$. If $U$ is an open subset of $X$, then the assignment
		\begin{equation*}
			\mathcal{F}|_{U}(V) = \mathcal{F}(V) \quad\text{for each open subset $V \subseteq U$}
		\end{equation*}
		define a presheaf on $U$. The presheaf $\mathcal{F}|_U$ is called the \inlinedef{restriction} of the presheaf $\mathcal{F}$ onto $U$.
	\end{example}
	
	\begin{example}[Sub-presheaf]
		Let $\mathcal{F}$ and $\mathcal{G}$ be sheaves on a topological space $X$ with values in algebraic objects such as Abelian groups or rings. If $\mathcal{F}(U)$ is a subgroup of $\mathcal{G}(U)$ for all open $U \subseteq X$, then $\mathcal{F}$ is said to be a \inlinedef{sub-presheaf} of $\mathcal{G}$.
	\end{example}
	
	By abuse of language we also say that $\mathcal{F}$ is a sub-presheaf of $\mathcal{G}$ if $\mathcal{F}$ is isomorphic to a sub-presheaf of $\mathcal{G}$.
	
	\subsection*{Sheaves}
	
	The fundamental property of functions on a topological space $X$ is that we can glue functions $f_i \colon U_i \to X$ defined on open subsets $U_i \subseteq X$ together into a function $f \colon U \to X$ defined on the union $U = \bigcup_i U_i$ of the open sets. Since we like to think of sections on $U$ as ``functions'' on $U$, the above property should hold for the type of presheaves which are desirable for our study. Hence we are led to the following definition.
	
	\begin{definition}
		Let $\mathcal{F}$ be a presheaf of Abelian groups on a topological space $X$. We say that $\mathcal{F}$ is a \inlinedef{sheaf} if it satisfies the following two properties: for all open subsets $U \subseteq X$ and for all open coverings $\Set{U_i}_{i \in I}$ of $U$,
		\begin{enumerate}
			\item (Uniqueness) Let $s, t \in \mathcal{F}(U)$ be sections such that $s|_{U_i} = t|_{U_i}$ for all $i \in I$. Then $s = t$.
			
			\item (Gluing) Given a collection of sections $s_i \in \mathcal{F}(U_i)$ such that $s_i|_{U_i \cap U_j} = s_j|_{U_i \cap U_j}$ for all $i, j \in I$, there exists $s \in\mathcal{F}(U)$ such that $s|_{U_i} = s_i$ for all $i \in I$.
		\end{enumerate}
	\end{definition} 
	
	The above two properties are called the \inlinedef{sheaf axioms}. Given a collection of sections $s_i \in \mathcal{F}(U_i)$ and $s \in\mathcal{F}(U)$ as in the second part, the section $s$ is said to be obtained by \inlinedef{gluing the sections} $s_i$. It is unique by the first part.
	
	Note that we may replace the target category from the category of Abelian groups to other categories, as noted in the previous section. For sheaves with values in a category of algebraic structures, the sheaf axioms can be restated in terms of exact sequences as follows.
	
	\begin{lemma}\label{chap3:sheaf-as-exact-sequences}
		Let $\mathcal{F}$ be a presheaf of Abelian groups (or rings, $R$-modules, $R$-algebras with $R$ a ring) on a topological space $X$. Then $\mathcal{F}$ is a sheaf if and only if for every open subset $U \subseteq X$ and open cover $\Set{U_i}_{i \in I}$ of $U$, the sequence
		\begin{equation*}
			0 \longrightarrow\mathcal{F}(U)\xlongrightarrow{\varphi} \prod_{i \in I} \mathcal{F}(U_i) \xlongrightarrow{\sigma} \prod_{(i, j) \in I \times I} \mathcal{F}(U_i \cap U_j),
		\end{equation*}
		is exact, where $\varphi\colon s \mapsto (s|_{U_i})_{i \in I}$ and $\sigma\colon (s_i)_{i \in I} \mapsto (s_i|_{U_i \cap U_j} - s_j|_{U_i \cap U_j})_{i, j}$.	
	\end{lemma}
	\begin{proof}
		The uniqueness condition is equivalent to the sequence being exact at $\mathcal{F}(U)$. Similarly, the gluing condition is equivalent to the sequence being exact at $\prod_{i} \mathcal{F}(U_i)$.
	\end{proof}
	
	A \inlinedef{morphism of sheaves} is a morphism of presheaves. We then obtain the category of sheaves on a topological space $X$, denoted by $(\Sheaf(X))$. Note that we really should have chosen a notation such as $(\Abs\Sheaf(X))$ to unambiguously clarify that it is the category of sheaves of \textit{Abelian groups} on $X$ that we are considering. Still, in this section and wherever appropriate, the proof works for any target category of sheaves. With this in mind, we shall prove most statement for sheaves of Abelian groups and left the other cases for the reader.
	
	The category of presheaves on $X$ is denoted by $(\PrSheaf(X))$. Clearly $(\Sheaf(X))$ is a full subcategory of $(\PrSheaf(X))$. 
	
	\begin{remark}
		For a sheaf $\mathcal{F}$ on a topological space $X$, $\mathcal{F}(\emptyset)$ is precisely the final object in the respective category. This can be seen from \cref{chap3:sheaf-as-exact-sequences} by considering the empty covering of the empty set. Since the empty product is the terminal object, we obtain the exact sequence $0 \to \mathcal{F}(\emptyset) \to 0 \to 0$, showing that $\mathcal{F}(\emptyset) \cong 0$.
		
		When $\mathcal{F}$ is a sheaf of sets, we can argue directly from the definition of sheaves that $\mathcal{F}(\emptyset)$ is a set with one element. Note that \cref{chap3:sheaf-as-exact-sequences} does not apply to this case.
	\end{remark}
	
	Sheaves can be restricted onto any open subset.
	
	\begin{example}[Restriction Sheaf]
		Suppose we have a sheaf $\mathcal{F}$ on a topological space $X$. If $U$ is an open subset of $X$, then the restriction $\mathcal{F}|_{U}$  is a sheaf.
	\end{example}
	
	\begin{example}[Subsheaf]
		Let $\mathcal{F}$ and $\mathcal{G}$ be sheaves on a topological space $X$ with values in algebraic objects such as Abelian groups or rings. If $\mathcal{F}(U)$ is a subgroup of $\mathcal{G}(U)$ for all open $U \subseteq X$, then $\mathcal{F}$ is said to be a \inlinedef{subsheaf} of $\mathcal{G}$.
	\end{example}
	
	It is not true in general that, if $\mathcal{G}$ is a sheaf, and $\mathcal{F}$ is a presheaf such that $\mathcal{F}(U)$ is a subgroup of $\mathcal{G}(U)$ for all $U \subseteq X$, then $\mathcal{F}$ is a subsheaf of $\mathcal{G}$. Hence a sub-presheaf of a sheaf is not necessarily a sheaf.
	
	\begin{example}[Sheaves of Continuous and Smooth Functions]
		Let $X$ and $Y$ be topological spaces. Define $\mathcal{C}$ to be the presheaf of continuous functions, that is, for every open subset $U \subseteq X$,
		\begin{equation*}
			\mathcal{C}(U) \coloneqq \Set*{f \colon U \to Y : \text{ $f$ is continuous}}.
		\end{equation*}
		The restriction maps are restrictions of functions. 
		
		In a similar manner, let $M$ be a smooth manifold. Then the presheaf of real-valued smooth functions on $M$ (\cref{chap3:presheaf-of-smooth-functions}) is a sheaf.
	\end{example}
	
	There are further examples of sheaves coming from differential geometry such as the sheaf of holomorphic function on $\CC$.
	
	\begin{example}[Hom-set as a Sheaf]
		Let $X$ be a topological space and $Y$ a set. Then the assignment $U \mapsto \Set{f \colon U \to Y : \text{$f$ is a function}}$ define a sheaf on $X$.
	\end{example}
	
	\begin{example}[Skyscraper Sheaves]
		Let $X$ be a topological space and $A$ an Abelian group. The \inlinedef{skyscraper sheaf in the point $p \in X$ with value $A$} is defined to be the sheaf
		\begin{equation*}
			\mathcal{F}(U) = \begin{cases}
				A,	& 	p \in U;\\
				0,	&	p \notin U.
			\end{cases}
		\end{equation*}
	\end{example}

	\subsection*{Sheaves on a Basis}
	
	Let $X$ be a topological space and $\mathcal{B}$ a basis of $X$. It will be useful, when we define affine schemes, to consider sheaves defined explicitly only on the basis $\mathcal{B}$. The main result of this section is that there is a unique sheaf on the whole topological space $X$ extending the sheaf defined on a basis.
	
	\begin{definition}
		Suppose $X$ is a topological space and $\mathcal{B}$ a basis of $X$.
		
		\begin{enumerate}
			\item A \inlinedef{presheaf on $\mathcal{B}$} is a contravariant functor $\mathcal{F}\colon \mathcal{B} \to (\Abs)$ where $\mathcal{B}$ is regarded as a full subcategory of $(\operatorname{Ouv}_X)$.
			
			\item A \inlinedef{sheaf on $\mathcal{B}$} is a presheaf $\mathcal{F}$ on $\mathcal{B}$ which satisfies the analogous conditions about gluing as for sheaves: for all $U \in\mathcal{B}$ and an open covering $U = \bigcup_{i \in I} U_i$ where $U_i \in \mathcal{B}$,
			\begin{enumerate}
				\item (Uniqueness) Let $s, t \in \mathcal{F}(U)$ be such that $s|_{U_i} = t|_{U_i}$ for all $i$. Then $s = t$.
				
				\item (Gluing) A collection of sections $s_i \in \mathcal{F}(U_i)$ is said to be \inlinedef{compatible} if for all $i, j \in I$, and all open set $U_k \in \mathcal{B}$ with $U_k \subseteq U_i \cap U_j$, we have
				\begin{equation*}
					s_i|_{U_k} = s_j|_{U_k}.
				\end{equation*}
				
				It is required that, whenever the sections $s_i \in \mathcal{F}(U_i)$ are compatible, then there is a section $s \in \mathcal{F}(U)$ such that $s|_{U_i} = s_i$. 
			\end{enumerate}
		\end{enumerate}
	\end{definition}
	
	Note that the gluing condition has to be modified from the gluing condition for sheaves. This is because the intersection of two basic open sets $U_i, U_j \in\mathcal{B}$ might not belong in $\mathcal{B}$. It is still open, however, and so we may cover the intersection $U_i \cap U_j$ by open sets from $\mathcal{B}$. So we need to check compatibility of sections in those smaller open sets instead.
	
	\begin{remark}
		If the basis $\mathcal{B}$ is stable under finite intersection (as is the case for the basis of spectra consisting of principal open subsets, \cref{chap2:principal-open-is-basis}), then the usual gluing condition can be used instead. For such a basis $\mathcal{B}$ , a presheaf $\mathcal{F}$ on $\mathcal{B}$ is a sheaf if for all $U \in\mathcal{B}$ and an open covering $U = \bigcup_{i \in I} U_i$ where $U_i \in \mathcal{B}$,
		\begin{enumerate}
			\item it satisfy uniqueness, and,
			\item for all sections $s_i \in \mathcal{F}(U_i)$, if $s_i|_{U_i \cap U_j} = s_j|_{U_i \cap U_j}$, then there is a section $s \in \mathcal{F}(U)$ such that $s|_{U_i} = s_i$.
		\end{enumerate}  
	\end{remark}

		Let $X$ be a topological space, $\mathcal{B}$ a basis of $X$, and $\mathcal{F}$ a sheaf on $\mathcal{B}$. There is a sheaf $\mathcal{F}'$ on $X$ such that $\mathcal{F}'(U) \cong \mathcal{F}(U)$ for all $U \in \mathcal{B}$. The sheaf $\mathcal{F}'$ is unique up to isomorphism.
	\end{proposition}

	\section{Stalks}
	
	Let $X$ be a topological space and let $x \in X$ be a point. The collection of Abelian groups $\mathcal{F}(U)$, where $U$  is an open neighbourhood of $x$, with restriction maps is a direct system. The direct limit of this system leads to the notion of stalks.
	
	\begin{definition}
		Let $\mathcal{F}$ be a presheaf on $X$. The \inlinedef{stalk} at $x \in X$ of $\mathcal{F}$ is the Abelian group
		\begin{equation*}
			\mathcal{F}_x \coloneq \varinjlim_{\substack{{U} \ni x, \\ {U} \text{ open}}} \mathcal{F}(U).
		\end{equation*}
	\end{definition}
	
	More explicitly, $\mathcal{F}_x$ is the set of equivalence classes of pairs $(U, s)$ where $U$ is an open neighbourhood of $x$ and $s \in \mathcal{F}(U)$. Two pairs $(U, s)$ and $(V, t)$ are definitionally equivalent if there is an open neighbourhood $W$ of $x$ such that $W \subseteq U \cap V$ and $s|_W = t|_W$. Let us denote the equivalence class of $(U, s)$ by $[U, s]$. The group operation on $\mathcal{F}_x$ is defined by
	\begin{equation*}
		[U, s] + [V, t] = [U \cap V, s|_{U\cap V} + t|_{U\cap V}].
	\end{equation*}
	
	\begin{exercise}
		Prove that the above relation is indeed an equivalence relation on the set of pairs $(U, x)$, and that the multiplication is well-defined which makes $\mathcal{F}_x$ into an Abelian group.
	\end{exercise}
	
	By the definition of direct limits, for each open neighbourhood $U$ of $x$ there is a natural homomorphism $\mathcal{F}(U) \to \mathcal{F}_x$ given by $s \mapsto [U, s]$. If $U$ is understood, we shall write $s_x$ for the equivalent class $[U, s]$. The element $s_x$ is called the \inlinedef{germ} of $s$ in $x$.
	
	If $\mathcal{F}$ and $\mathcal{G}$ are presheaves on $X$ with a morphism $\varphi \colon \mathcal{F} \to \mathcal{G}$, then there is an induced homomorphism of Abelian group between stalks
	\begin{align*}
		\varphi_x \colon \mathcal{F}_x & \longrightarrow \mathcal{G}_x \\
	[U, s] & \longmapsto [U, \varphi_U(s)].
	\end{align*}
	This is the same map from the universal property of direct limits. 
	
	\begin{exercise}
		Show that $\varphi_x$ is a group homomorphism.
	\end{exercise}
	
	We thus see that taking stalks gives a functor from the category of presheaves on $X$ to the category of Abelian groups. Further, if the presheaves take values in sets or rings, then stalks will be sets or rings, respectively. 
	
	\begin{lemma}\label{chap3:germs-equal-implies-equal}
		Let $\mathcal{F}$ be a sheaf on a topological space $X$ and let $s, t \in\mathcal{F}(X)$ be sections such that $s_x = t_x$ for all $x \in X$. Then $s = t$.
	\end{lemma}
	\begin{proof}
		For each $x \in X$, there is an open neighbourhood $U_x$ such that $s|_{U_x} = t|_{U_x}$. Since the open sets of the form $U_x$ covers $X$, this shows that $s = t$.
	\end{proof}
	
	Injectivity and bijectivity of morphisms in the data of a morphism of sheaves can be checked on stalks, as the following proposition shows.
	
	\begin{proposition}\label{chap3:injective-can-be-checked-on-stalks}
		Let $X$ be a topological space, $\mathcal{F}$ and $\mathcal{G}$ sheaves on $X$, and $\varphi, \psi \colon \mathcal{F} \to \mathcal{G}$ morphisms of sheaves. Then
		
		\begin{enumerate}
			\item The homomorphisms $\varphi_U \colon \mathcal{F}(U) \to \mathcal{G}(U)$ are injective for all open $U \subseteq X$ if and only if the induced morphisms $\varphi_x \colon \mathcal{F}_x \to \mathcal{G}_x$ are injective for all $x \in X$.
			
			\item The homomorphisms $\varphi_U \colon \mathcal{F}(U) \to \mathcal{G}(U)$ are bijective for all open $U \subseteq X$ if and only if the induced morphisms $\varphi_x \colon \mathcal{F}_x \to \mathcal{G}_x$ are bijective for all $x \in X$.
			
			\item $\varphi_x = \psi_x$ for all $x \in X$ if and only if $\varphi = \psi$.
		\end{enumerate}
	\end{proposition}
	
	\begin{proof}
		We shall prove the statements (2.) and (3.), leaving (1.) as an exercise. 
		
		If the homomorphisms $\varphi_U$ are bijective for all open subset $U$, then the stalks are isomorphic under $\varphi_x$ by the universal property of direct limits. (Abstract nonsense: every functor preserves isomorphisms.) So we only have to prove the converse. Assume that $\varphi_x \colon \mathcal{F}_x \to \mathcal{G}_x$ are bijective for all $x \in X$. 
		
		(Injectivity.) Let $s, t \in \mathcal{F}(U)$ be sections and suppose that $\varphi_U(s) = \varphi_U(t)$. Then for every $x \in U$, we have $\varphi_x(s_x) = [U, \varphi_U(s)] = [U, \varphi_U(t)] = \varphi_x(t_x)$. Hence $s_x = t_x$ for all $x \in U$, and so $s = t$ by \cref{chap3:germs-equal-implies-equal}.
		
		(Surjectivity.) Let $b \in \mathcal{G}(U)$. For each $x \in X$, there is a germ $s_x \in \mathcal{F}_x$ such that $\varphi_x(s_x) = b_x$. Let $s$ be a section defined on an open neighbourhood $V$ of $x$ such that $[V, s] = s_x$. Then by assumption $[V, \varphi_V(s)] = b_x$. By definition, there is an open neighbourhood $W_x$ of $x$ and a section $w(x) \coloneqq s|_{W_x} \in \mathcal{F}(W_x)$ such that $\varphi_{W_x}(w(x)) = b|_{W_x}$.
		
		The open sets $\Set{W_x}$, $x \in U$, form an open cover of $U$. Furthermore, if $x, y\in U$, then $w(x)|_{W_x \cap W_y} = w(y)|_{W_x \cap W_y}$. Indeed, we have
		\begin{align*}
			\varphi_{W_x \cap W_y}(w(x)|_{W_x \cap W_y}) & = \varphi_{W_x}(w(x))|_{W_x \cap W_y} && \text{(Definition of morphism of sheaves)}\\
			& = b|_{W_x}|_{W_x \cap W_y} && \text{(Definition of $w(x)$)} \\
			& = b|_{W_y}|_{W_x \cap W_y} && \text{(Definition of morphism of sheaves)}\\
			& = \varphi_{W_y}(w(y))|_{W_x \cap W_y} && \text{(Definition of $w(y)$)}\\
			& = \varphi_{W_x \cap W_y}(w(y)|_{W_x \cap W_y}) && \text{(Definition of morphism of sheaves)}.
		\end{align*}
		Since we have just proved that the morphism $\varphi_{W_x \cap W_y}$ is injective, and thus $w(x)|_{W_x \cap W_y} = w(y)|_{W_x \cap W_y}$ as claimed. In particular, the sections $w(x)$ can be glued together. Let $w \in \mathcal{F}(U)$ denote the resulting section so that $w|_{W_x} = w(x)$ for all $x \in U$. Then
		\begin{equation*}
			\varphi_U(w)|_{W_x} = \varphi_{W_x}(w|_{W_x}) = \varphi_{W_x}(w(x)) = b|_{W_x}
		\end{equation*}
		for all open set $W_x$ in the cover. By the sheaf condition, $\varphi_U(w) = b$.
		
		Finally we prove that if $\varphi_x = \psi_x$ for all $x \in X$, then $\varphi = \psi$. Note that the converse is trivial. Let $s \in \mathcal{F}(U)$ be a section. Then
		\begin{equation*}
			(\varphi_U(s))_x =\varphi_x(s_x) = \psi_x(s_x) = (\psi_U(s))_x
		\end{equation*}
		 for all $x \in X$. Hence $\varphi_U(s) = \psi_U(s)$ by \cref{chap3:germs-equal-implies-equal}.
	\end{proof}
	
	\begin{exercise}
		Finish the proof of \cref{chap3:injective-can-be-checked-on-stalks}.
	\end{exercise}
	
	\begin{remark}
		Keep the notation of \cref{chap3:injective-can-be-checked-on-stalks}.
		
		\begin{enumerate}
			\item Note that this proof does not use the fact that $\mathcal{F}$, $\mathcal{G}$ is a sheaf of Abelian groups, so the proposition holds for sheaves of sets.
			
			\item The analogous statement for surjections is false. It is in general not true that $\varphi_U \colon \mathcal{F}(U) \to \mathcal{G}(U)$ are surjective for all open $U \subseteq X$ if and only if the induced morphisms $\varphi_x \colon \mathcal{F}_x \to \mathcal{G}_x$ are surjective for all $x \in X$.
		\end{enumerate}
	\end{remark}
	
	The following proposition is much harder to prove.
	
	\begin{proposition}\label{chap3:morphisms-in-term-of-stalks}
		Let $X$ be a topological space, $\mathcal{F}$, $\mathcal{G}$ sheaves on $X$, and $\varphi \colon \mathcal{F} \to \mathcal{G}$ a morphism of sheaves. Then
		\begin{enumerate}
			\item $\varphi$ is a monomorphism in $(\Abs(X))$ if and only if $\varphi_x$ is injective for all $x \in X$.
			
			\item $\varphi$ is an epimorphism in $(\Abs(X))$ if and only if $\varphi_x$ is surjective for all $x \in X$.
			
			\item $\varphi$ is an isomorphism in $(\Abs(X))$ if and only if $\varphi_x$ is bijective for all $x \in X$.
		\end{enumerate}
	\end{proposition}
	\begin{proof}
		(1) First assume that $\varphi$ is a monomorphism. In light of \cref{chap3:injective-can-be-checked-on-stalks}, it suffices to show that each morphism $\varphi_U \colon \mathcal{F} \to \mathcal{G}$ is injective. Let $\mathcal{K}$ be the kernel sheaf of $\varphi$, i.e. $\mathcal{K}(U) = \ker\varphi_U$ for all open $U\subseteq X$. This is a subsheaf of $\mathcal{F}$ (\cref{chap3:kernel-sheaf-is-a-sheaf}).
		
		Let $i \colon \mathcal{K} \to \mathcal{F}$ be the inclusion morphism, i.e. $i_U$ is the inclusion $\mathcal{K}(U) \hookrightarrow \mathcal{F}(U)$ for all open subset $U \subseteq X$. There is also another morphism of sheaves $0 \colon \mathcal{K} \to \mathcal{F}$ such that $0_U \colon \mathcal{K}(U) \to \mathcal{F}(U)$ is the zero homomorphism for all $U$. Clearly $\varphi \circ i = 0 = \varphi \circ 0$. Since $\varphi$ is a monomorphism, $i = 0$. Hence for each open $U \subseteq X$, we see that $a = i_U(a) = 0_U(a) = 0$ for all $a \in \ker(\varphi_U)$. In other words, $\varphi_U$ is injective for all open subset $U$.
		
		Conversely, assume that $\varphi_x$ is injective for all $x \in X$. Suppose $\psi, \eta \colon\mathcal{H} \to \mathcal{F}$ are morphisms of sheaves such that $\varphi \circ \psi = \varphi \circ \eta$. Since taking stalks is a functor, $\varphi_x \circ \psi_x = \varphi_x \circ \eta_x$ for all $x \in X$. Using the fact that $\varphi_x$ is injective, we have $\psi_x = \eta_x$ for all $x \in X$. By \cref{chap3:injective-can-be-checked-on-stalks}, $\psi = \eta$, and so $\varphi$ is a monomorphism.

	\end{proof}

	\section{Sheafification}
	
	As seen above, some presheaves fail to be sheaves because of the failure to glue sections.  There is a canonical way to associate a sheaf to a presheaf.
	
	\begin{theorem}[Sheafification]\label{chap3:sheafification}
		Let $X$ be a topological space and $\mathcal{F}$ a presheaf. There exists a sheaf $\mathcal{F}^\dagger$ (called the \inlinedef{sheafification} of $\mathcal{F}$) with a morphism of presheaves $i \colon \mathcal{F} \to \mathcal{F}^\dagger$ such that if $\mathcal{G}$ is a sheaf with a morphism of presheaves $\varphi \colon \mathcal{F} \to \mathcal{G}$, then there is a unique morphism of sheaves ${\varphi}^\dagger \colon \mathcal{F}^\dagger \to \mathcal{G}$ such that the following diagram commutes:

		\[\begin{tikzcd}
			{\mathcal{F}} & {\mathcal{F}^\dagger} \\
			& {\mathcal{G}}
			\arrow["i", from=1-1, to=1-2]
			\arrow["\varphi"', from=1-1, to=2-2]
			\arrow["{\varphi^\dagger}", dashed, from=1-2, to=2-2]
		\end{tikzcd}\]
		
		The pair $(\mathcal{F}^\dagger, i)$ is unique up to unique isomorphism. Moreover, $i_x \colon \mathcal{F}_x \to \mathcal{F}^\dagger_x$ is an isomorphism for all $x \in X$.
	\end{theorem}
	
	The sheaf $\mathcal{F}^\dagger$ is also called the \inlinedef{sheaf associated to the presheaf} $\mathcal{F}$.
	
	\begin{proof}
	We divide the proof into several parts.
	
	\textbf{[Definition.]} Let $U \subseteq X$ be an open subset. Define $\mathcal{F}^\dagger(U)$ to be the set of functions $f \colon U \to \coprod_{x \in U} \mathcal{F}_x$ satisfying
		\begin{itemize}
			\item [(\ast)] for all $x \in U$, there is an open neighbourhood $V \subseteq U$ of $x$ and $s \in \mathcal{F}(V)$ such that for all $y \in V$, $f(y) = s_y$.
		\end{itemize} 
	With pointwise addition this is an abelian group. The restriction map $\res^U_W \colon \mathcal{F}^\dagger(U) \to \mathcal{F}^\dagger(W)$ for $W \subseteq U$ is defined to be the restriction of functions, so $\res^U_W(f) = f|_W$ for all $f \in \mathcal{F}^\dagger(U)$. It is easily checked that $f|_W \in \mathcal{F}^\dagger(W)$, and that restrictions are group homomorphisms. From this it follows that $\mathcal{F}^\dagger$ is a presheaf on $X$.
	
	\textbf{[$\mathcal{F}^\dagger$ is a sheaf.]} To show that $\mathcal{F}^\dagger$ is a sheaf, we assume that $U \subseteq X$ is an open set with an open covering $\bigcup_{i \in I}U_i = U$.
	
	\begin{enumerate}
		\item (Uniqueness) Easy.
		
		\item (Gluing.) For each $i \in I$, let $f_i \in \mathcal{F}^\dagger(U_i)$ be functions such that $f_i|_{U_i \cap U_j} = f_j|_{U_i \cap U_j}$ for all $i, j \in I$. Define $f \colon U \to \coprod_{x \in U} \mathcal{F}_x$ by $f(x) = f_i(x)$ if $U_i$ contains $x$. This is well-defined because if $U_i$ and $U_j$ both contains $x$, then $x \in U_i \cap U_j$ and so $f_i(x) = f_i|_{U_i \cap U_j}(x) = f_j|_{U_i \cap U_j} = f_j(x)$. Further, the condition (\ast) hold for $f$.
	\end{enumerate}
	
	\textbf{[Definition of $i$.]} Next we define a morphism of presheaves $i \colon \mathcal{F} \to \mathcal{F}^\dagger$. For each $a \in \mathcal{F}(U)$, let $i_U(a)$ be the function $x \mapsto a_x$ for all $x \in U$. This is clearly seen to be a group homomorphism compatible with restriction maps of $\mathcal{F}$ and $\mathcal{F}^\dagger$. Thus $i$ is really a morphism of presheaves.
	
	\textbf{[$i_x$ is surjective.]} Let $x \in X$ We now show that $\mathcal{F}^\dagger_x \cong \mathcal{F}_x$ under the map $i_x$. Let $U$ be an open neighbourhood of $x$ and let $f \in \mathcal{F}^\dagger(U)$. By definition there is an open neighbourhood $V$ of $x$ and $s \in \mathcal{F}(V)$ such that for all $y \in V$, we have $f(y) = s_y$. Then $[U, f] = [V, i_V(s)]$. In particular, the map $i_x \colon \mathcal{F}_x \to \mathcal{F}^\dagger_x$ is a surjection. 
	
	\textbf{[$i_x$ is injective.]} To show that $i_x$ is injective, let $0$ denote the function $x \mapsto 0_x$ for all $x \in X$. Plainly, $[X, 0]$ is the additive identity of $\mathcal{F}_x$. Suppose that $V$ is a neighbourhood of $x$ and $s \in \mathcal{F}(V)$ is a section  such that $i_x([V,s ]) = [V, i_V(s)] = [X, 0]$. By definition, there is an open neighbourhood $W \subseteq V$ of $x$ such that $i_W(s|W) = i_V(s)|_W = 0|_W$. But this implies that $s_y = 0_y$ for all $y \in W$. By \cref{chap3:germs-equal-implies-equal}, $s|_W = 0$. Hence $[V, s] = [W, s|_W] = [W, 0]$, and so $i_x$ is an isomorphism.

	\section{Direct Image and Inverse Image}
	
	Suppose we have a continuous map $f \colon X \to Y$ between two topological spaces. There is a way to transfer a sheaf on $X$ to a sheaf on $Y$, and vice versa. We first start with the notion of pushing a sheaf on $X$ by $f$ into a sheaf on $Y$.
	
	\begin{definition}[Pushforward, Direct Image Sheaf]
		Let $f \colon X \to Y$ be a continuous map of topological spaces. If $\mathcal{F}$ is a sheaf on $X$, then the \inlinedef{pushforward of $\mathcal{F}$ under $f$} or the \inlinedef{direct image sheaf of $\mathcal{F}$} is the sheaf $f_\ast\mathcal{F}$ defined by
		\begin{equation*}
			f_\ast\mathcal{F}(V) = \mathcal{F}(f^{-1}(V))
		\end{equation*}
		for all open subset $V \subseteq Y$. The restriction maps of $f_\ast\mathcal{F}$ is the restriction maps of $\mathcal{F}$.
	\end{definition}
	
	\begin{exercise}~
		\begin{enumerate}
			\item Prove that if $\mathcal{F}$ is a presheaf on $X$, then $f_\ast\mathcal{F}$ defined as above is a presheaf on $Y$.
			
			\item Prove that if $\mathcal{F}$ is a sheaf, then $f_\ast\mathcal{F}$ is also a sheaf on $Y$.
		\end{enumerate}
	\end{exercise}
	
	\begin{exercise}
		Prove that $f_\ast$ define a functor $(\Sheaf(X)) \to (\Sheaf(Y))$.
	\end{exercise}
	
	On the other hand, pulling back a sheaf on $Y$ to a sheaf on $X$ is harder. This is due to the fact that the image $f(U)$ of an open set $U\subseteq X$ under a continuous map need not be open. 	\begin{equation*}

	\end{equation*}

	\subsection*{Ringed Spaces}
	
	\begin{definition}
		A \inlinedef{ringed space} is a pair $(X, \mathcal{O}_X)$ where $X$ is a topological space and $\mathcal{O}_X$ is a sheaf of rings on $X$.
	\end{definition}
	
	The sheaf $\mathcal{O}_X$ is called the \inlinedef{structure sheaf} of $X$. A \inlinedef{morphism of ringed spaces} $(f, f^\flat) \colon (X, \mathcal{O}_X) \to (Y,\mathcal{O}_Y)$ is a pair consisting of a continuous function $f \colon X \to Y$ and a morphism $f^\flat \colon \mathcal{O}_Y \to f_\ast\mathcal{O}_Y$ of sheaves of rings. 
	
	\begin{exercise}
		Define the composition of two morphisms of ringed spaces. Use this to show that ringed spaces and their morphisms form a category of ringed spaces.
	\end{exercise}
	
	\subsection*{Locally Ringed Spaces}
	
	Recall that a \inlinedef{local ring} is a ring $A$ with a unique maximal ideal. We write $\mathfrak{m}_A$ for the maximal ideal of $A$, and define the \inlinedef{residue class field} of $A$ to be $\kappa(A) = A/\mathfrak{m}_A$. A ring homomorphism $\varphi \colon A \to B$ between local rings is said to be \inlinedef{local} if $\varphi(\mathfrak{m}_A) \subseteq \mathfrak{m}_B$, or, equivalently, if $\varphi^{-1}(\mathfrak{m}_B) = \mathfrak{m}_A$.
	
	\begin{definition}~
		\begin{enumerate}
			\item A \inlinedef{locally ringed space} is a ringed space $(X, \mathcal{O}_X)$ such that for all $x \in X$ the stalk $\mathcal{O}_{X, x}$ is a local ring. The ring $\mathcal{O}_{X, x}$ is called the \inlinedef{local ring of $X$ in $x$}.
			
			\item A \inlinedef{morphism of locally ringed spaces} $(f, f^\flat) \colon (X, \mathcal{O}_X) \to (Y, \mathcal{O}_Y)$ is a morphism of ringed spaces such that the induced map of stalks
			\begin{equation*}
				f^\sharp_x \colon (f^{-1}\mathcal{O}_Y)_x = \mathcal{O}_{Y, f(x)} \to \mathcal{O}_{X, x}
			\end{equation*}
			is a local ring homomorphism for all $x \in X$.
		\end{enumerate}	
	\end{definition}

	Let $(X, \mathcal{O}_X)$ be a locally ringed space, $x \in X$, and $\kappa(x)$ the residue field at $x$. If $f \in\mathcal{O}_X(U)$ is a section defined on an open neighbourhood $U$ of $x$, we define $f(x)$ to be the image of $f$ under the composition
	\begin{equation*}
		\mathcal{O}_X(U) \to \mathcal{O}_{X, x} \to \mathcal{O}_{X, x}/\mathfrak{m}_x = \kappa(x).
	\end{equation*}
	In this manner we may think of a section on $U$ as a ``function'' from $U$ whose value at each point lies in the different fields $\kappa(x)$.
	
	\begin{lemma}\label{chap3:set-where-section-is-non-zero-is-closed}
		Let $(X, \mathcal{O}_X)$ be a locally ringed space, and let $f \in \mathcal{O}_X(X)$. Define 
		\begin{equation*}
			X_f \coloneqq \Set*{x \in X : f(x) \neq 0}.
		\end{equation*}
		Then $X_f$ is an open subset of $X$.
	\end{lemma}
	\begin{proof}
		Let $x \in X_f$ and thus $f_x \notin \mathfrak{m}_x$ by unravelling the definitions. Since $\mathcal{O}_{X, x}$ is a local ring, this means that $f_x$ is invertible in $\mathcal{O}_{X, x}$. So there is $g_x \in \mathcal{O}_{X, x}$ such that $f_x g_x = 1$. Let $g \in\mathcal{O}_X(U)$ be a lift of $g_x$ to an open neighbourhood $U$ of $x$. Therefore there is an open neighbourhood $V \subseteq U$ such that
		\begin{equation*}
			f|_V g|_V = 1.
		\end{equation*}
		This equation implies that $f_y$ is invertible for all $y \in V$, and so $f(y) \neq 0$ for all $y \in V$. Hence $x \in V \subseteq X_f$ and thus the set $X_f$ is open.
	\end{proof}
	
	Compare this to the case of affine varieties.
	
	\section{Affine Schemes}
	
	Let $A$ be a ring. We are going to make $X = \Spec A$ into a locally ringed space by constructing a sheaf of rings on $X$. Recall that $X$ has a basis consisting of principal open subsets. Define a sheaf $\mathcal{O}_X$ on this basis by setting
	\begin{equation*}
		\mathcal{O}_X(D(f)) = A_f\quad\text{for each $ f\in A$},
	\end{equation*}
	where $A_f$ is the localisation of $A$ with respect to the set $\Set{f^i : i \geq 0}$. Let us denote the canonical homomorphism $A \to A_f$ by $i_f$. To see that this is well-defined, let $D(f) = D(g)$. By \cref{chap2:covering-trick}, $D(g) \subseteq D(f)$ if and only $g^n =  fa$ for some $a \in A$ and $n \geq 1$. In particular, $f/1$ is invertible in $A_g$ because $(a/g^n)(f/1) = 1$. By the universal property of localisation, there exists a unique ring homomorphism $\rho^f_g \colon A_f \to A_g$ such that $\rho^f_g\circ i_f = i_g$. In fact, the homomorphism $\rho^f_g$ is a homomorphism of $R$-algebras.
	
	Suppose $D(h) \subseteq D(g) \subseteq D(f)$. Then by uniqueness, $\rho^g_h \circ \rho^f_g = \rho^f_h$. In particular, if $D(g) = D(f)$, then $\rho^f_g$ is an isomorphism which allows us to identify $A_f$ with $A_g$. The condition $\rho^g_h \circ \rho^f_g = \rho^f_h$ implies that $\mathcal{O}_X$ is a presheaf of rings on $\mathcal{B}$ with the restriction maps given by $\rho^f_g$.
	
	\begin{theorem}
		The presheaf $\mathcal{O}_X$ is a sheaf on $\mathcal{B}$.
	\end{theorem}
	\begin{proof}
		Let $D(f)$ be a principal open subset and let $\Set{D(f_i) : i \in I}$ be an open cover of $D(f)$. By restricting the presheaf $\mathcal{O}_X$ to $D(f)$, and replacing $A_f$ with $A$, we may assume that $f = 1$, and hence $D(f) = X$, to ease the notation. Therefore the assumption translates to the statement that $X = \Spec A$ is covered by $\Set{D(f_i) : i \in I}$.
		
		We first prove the sheaf conditions when $I$ is finite.
		
		\begin{enumerate}
			\item Let $s \in \mathcal{O}_X(X) = A$ satisfy $s|_{D(f_i)} = 0$ for all $i \in I$. For each $i$ there is $n_i \in \NN$ such that $f_i^{n_i}s = 0$. Set $n = \max_{i \in I} n_i$, so that $f_i^n s = 0$.
			
			Since $D(f_i) = D(f^n_i)$ for all $i$, it follows that $\bigcup_{i \in I}D(f_i^n) = \bigcup_{i \in I}D(f_i) = X$ and so $\sum_{i \in I}f^n_i A = A$. Thus there are $b_i \in A$ such that $\sum_{i \in I} f^n_i b_i = 1$. This gives
			\begin{equation*}
				s = \left(\sum_{i \in I} f^n_i b_i\right)s = \sum_{i \in I} f^n_i b_i s = 0.
			\end{equation*}
			
			\item Let $s_i \in \mathcal{O}_X(D(f_i))$ satisfy $s_i|_{D(f_i) \cap D(f_j)} = s_j|_{D(f_i) \cap D(f_j)}$ for all $i, j \in I$. Let us write $s_i = \frac{a_i}{f_i^{n_i}}$. We may adjust the fractions so that all $n_i$ are equal, so that $s_i = \frac{a_i}{f_i^{n}}$, where $n \in \NN$ is independent of $i$.
			
			By hypothesis, for all $i, j \in I$,
			\begin{equation*}
			\left.\frac{a_i}{f_i^n}\right\vert_{D(f_if_j)}= \frac{a_if_j^n}{(f_if_j)^n} = \frac{a_jf_i^n}{(f_if_j)^n} = \left.\frac{a_j}{f_j^n}\right\vert_{D(f_if_j)}.
			\end{equation*}
			So that there is $m \geq 1$ such that $(f_if_j)^m(a_if_j^n - a_jf_i^n) = 0$. Again we may choose $m$ to be independent of $i$ and $j$. Thus by setting $k = n + m$, we obtain
			\begin{equation*}
				(f_i^m a_i) f_j^k = (f_j^ma_j)f_i^k
			\end{equation*}
			for all $i, j \in I$.
			
			Arguing as above, there are $b_i \in A$ such that $\sum_{j \in I} f^k_j b_j = 1$. Put $s \coloneqq\sum_{j \in I} b_j f_j^m a_j$. Then
			\begin{equation*}
				f_i^ks = f_i^k \left(\sum_{j \in I} b_j f_j^m a_j\right) = \sum_{j \in I} b_j (f_i^k f_j^m a_j) = \sum_{j \in I} b_j (f_i^m a_i f_j^k) = f_i^m a_i \left(\sum_{j \in I} b_j  f_j^k\right) = f_i^m a_i.
			\end{equation*}
			
			Hence $s \in A = \mathcal{O}_X(X)$ is such that its image in $A_{f_i}$ is equal to
			\begin{equation*}
				\frac{s}{1} = \frac{f_i^k s}{f_i^k} = \frac{f_i^m a_i}{f_i^{n + m}} = \frac{a_i}{f_i^n} = s_i.
			\end{equation*}
		\end{enumerate}
		
		Finally we consider the case when $I$ is infinite. Via \cref{chap2:covering-trick}, there is a finite subset $J \subseteq I$ such that $1 = \sum_{j \in J} f_j A$. From what was just proved, we see that gluing sections indexed by finite subsets work.
		
		\begin{enumerate}
			\item Let $s \in \mathcal{O}_X(X) = A$ satisfy $s|_{D(f_i)} = 0$ for all $i \in I$. Then $s|_{D(f_j)} = 0$ for all $j \in J$ and hence $s = 0$.
			
			\item Let $s_i \in \mathcal{O}_X(D(f_i))$ satisfy $s_i|_{D(f_i) \cap D(f_k)} = s_k|_{D(f_i) \cap D(f_k)}$ for all $i, k \in I$. Consider the sections $s_j$ with $j \in J$. From the finite case, they glue to a section $s \in \mathcal{O}_X(X)$ such that $s|_{D(f_j)} = s_j$ for all $j \in J$. We claim that the section $s$ is the one we get from gluing $s_i$.
			
			We only need to check that for all $i \in I$, $s|_{D(f_i)} = s_i$. Note that $\Set{D(f_i) \cap D(f_j)}_{j \in J}$ is a finite covering of $D(f_i)$. Hence, for all $i \in I$ and $j \in J$,
			\begin{equation*}
				s|_{D(f_i)}|_{D(f_i) \cap D(f_j)} = s|_{D(f_j)}|_{D(f_i) \cap D(f_j)} = s_j|_{D(f_i) \cap D(f_j)} = s_i|_{D(f_i) \cap D(f_j)}.
			\end{equation*}
			By (1), $s|_{D(f_i)} = s_i$. This finishes the proof. \qedhere
		\end{enumerate}
	\end{proof}
	
	\begin{corollary}
		The sheaf $\mathcal{O}_X$ extends to a sheaf $\mathcal{O}_X$ on $X$, called the \inlinedef{structure sheaf} of $\Spec A$.
	\end{corollary}
	
	\begin{lemma}[Stalks of $\Spec A$] 
		Let $X = \Spec A$, $\mathcal{O}_X$ its structure sheaf, and $x \in \Spec A$. Then		
		\begin{equation*}
			\mathcal{O}_{X, x} = A_{\mathfrak{p}_x},
		\end{equation*}
		the localisation of $A$ at the prime ideal $\mathfrak{p}_x$. In particular, $(X, \mathcal{O}_X)$ is a locally ringed space.
	\end{lemma}
	\begin{proof}
		By \cref{chap3:stalks-can-be-computed-on-basis}, we can compute the stalk $\mathcal{O}_{X, x}$ on the basis of principal open subsets, and thus
		\begin{equation*}
			\mathcal{O}_{X, x} = \varinjlim_{D(f) \ni x} \mathcal{O}_X(D(f)) = \varinjlim_{f \notin\mathfrak{p}} A_f = A_{\mathfrak{p}},
		\end{equation*}
		the last equality follows from commutative algebra (\cref{chap3:localisation-at-prime-as-direct-limit}).
	\end{proof}
	
	\begin{definition}
		A locally ringed space $(X, \mathcal{O}_X)$ is said to be an \inlinedef{affine scheme} if it is isomorphic to a locally ringed space $(\Spec A, \mathcal{O}_{\Spec A})$ for suitable ring $A$.
	\end{definition}
	
	A \inlinedef{morphism of affine schemes} is simply a morphism of locally ringed spaces. We thus obtain the category of affine schemes which we shall denote by $(\Aff)$.
	
	Let $\varphi \colon A \to B$ be a ring homomorphism. We shall define a morphism of affine schemes $\Spec \varphi = (f, f^\flat) \colon (\Spec B, \mathcal{O}_{\Spec B}) \to(\Spec A, \mathcal{O}_{\Spec A})$. Here is the definition: 
	\begin{enumerate}
		\item $f = \Spec \varphi$ is the induced continuous map on spectra, and,
		\item $f^\flat \colon \mathcal{O}_{\Spec A} \to f_\ast \mathcal{O}_{\Spec B}$ is given on principal open subsets by  
		\begin{equation*}
			f^\flat_{D(s)} : \mathcal{O}_{\Spec A}(D(s)) \to \mathcal{O}_{\Spec B}(f^{-1}(D(s))).
		\end{equation*}
		Since $f^{-1}(D(s)) = D(\varphi(s))$ by \cref{chap2:property-of-spec-map}, we need to define a map $f^\flat_{D(s)} \colon A_s \to B_{\varphi(s)}$. The natural map for this is the unique map $\varphi_s \colon A_s \to B_{\varphi(s)}$ induced by the canonical localisation maps:

	\section{Schemes}
	
	After defining affine schemes, we can now define schemes as locally ringed spaces that are affine schemes glued together.
	
	\begin{definition}
		A \inlinedef{scheme} is a locally ringed space $(X, \mathcal{O}_X)$ which admits a covering by open subsets $\Set{U_i}_{i \in I}$ such that $(U_i, \mathcal{O}_{X | U_i})$ is isomorphic to an affine scheme for all $i \in I$.
	\end{definition}
	
	A \inlinedef{morphism of schemes} is a morphism of the underlying locally ringed spaces. Schemes and morphisms of schemes together form the category of schemes, which is denoted by $(\Schs)$. The category of affine schemes is then a full subcategory of $(\Schs)$. We shall often write $X$ for the scheme $(X, \mathcal{O}_X)$; its structure sheaf is left implicit.
	
	\subsection*{Open Subschemes}
	
	Open subsets of a scheme inherit the scheme structure of $X$, as the following lemma shows.
	
	\begin{lemma}
		Let $(X, \mathcal{O}_X)$ be a scheme and $U \subseteq X$ an open subset. The locally ringed space $(U, \mathcal{O}_{X|U})$ is itself a scheme. 
	\end{lemma}
	\begin{proof}
		It is clear that $(U, \mathcal{O}_{X|U})$ is a locally ringed space. Let $\Set{U_i}_{i \in I}$ be an open covering of $X$ such that each $(U_i, \mathcal{O}_{X|U_i})$ is isomorphic to an affine scheme. For each $i \in I$, $U \cap U_i$ is an open subset of $U_i$. Using the fact that $U_i$ is affine, we see that there is an open covering of $U \cap U_i$ by principal open subsets $D(f_{i,j}), {j \in J}$. By \cref{chap3:principal-open-subset-is-affine}, $D(f_{i,j}) \cong \Spec \Gamma(U_i, \mathcal{O}_X)_{f_{i,j}}$ as affine schemes, and so $U \cap U_i$ can be covered by affine schemes. It then follows that $U = \bigcup_i (U \cap U_i)$ can also be covered by affine schemes.
	\end{proof}
	
	\begin{definition}
		Let $X$ be a scheme and $U \subseteq X$ an open subset. The scheme $(U, \mathcal{O}_{X|U})$ is called an \inlinedef{open subscheme} of $X$. If, in addition, $U$ is affine, we say that $U$ is an \inlinedef{affine open subscheme} of $X$.
	\end{definition}
	
	The proof of the lemma shows that every scheme has a basis consisting of affine open subschemes. We also record another useful lemma regarding sections of schemes. It says that non-empty open subsets of a scheme cannot have zero sections.
	
	\begin{lemma}\label{chap4:scheme-empty-section-iff-empty}
		Let $(X, \mathcal{O}_X)$ be a scheme and $U \subseteq X$ an open subset. Then $\Gamma(U, \mathcal{O}_X) = 0$ if and only if $U = \emptyset$.
	\end{lemma}
	\begin{proof}
		If $U = \emptyset$, then $\Gamma(U, \mathcal{O}_X) = 0$ which can be seen from the sheaf conditions with respect to the empty covering.
		
		Now suppose $\Gamma(U, \mathcal{O}_X) = 0$. Let $V \subseteq U$ be an affine open subscheme of $X$, which exists because affine open subschemes form a basis of the topology of a scheme. Then the restriction map is a ring homomorphism
		\begin{equation*}
			0 = \Gamma(U, \mathcal{O}_X) \to \Gamma(V, \mathcal{O}_X),
		\end{equation*}
		which implies that $\Gamma(V, \mathcal{O}_X) = 0$. Therefore $V = \Spec 0 = \emptyset$ by \cref{chap3:antiequivalence-of-affine-schemes-and-rings}. Taking a covering of $U$ by affine open subschemes, each of which is then the empty set, we conclude that $U = \emptyset$.
	\end{proof}
	
	Let $X$ be a scheme and $U \subseteq X$ an open subset. Then the inclusion $j \colon U \hookrightarrow X$ gives rise to a morphism of schemes $(U, \mathcal{O}_{X|U}) \to (X, \mathcal{O}_X)$ whose morphism of sheaves $\mathcal{O}_X \mapsto j_\ast \mathcal{O}_{X|U}$ we shall shortly define. Let $V \subseteq U$ be an open set. The restriction maps of the structure sheaf $\mathcal{O}_X$ give us the ring homomorphism
	\begin{equation*}
		\Gamma(V, \mathcal{O}_X) \to \Gamma(V \cap U, \mathcal{O}_X) = \Gamma(j^{-1}(V), \mathcal{O}_{X|U}) = \Gamma(V, j_\ast\mathcal{O}_{X|U}).
	\end{equation*}
	This is compatible with restriction $W \subseteq V \subseteq U$, and thus define a morphism of sheaves $j^\flat\colon \mathcal{O}_X \to j_\ast\mathcal{O}_{X|U}$.
	
	The following lemma will be useful for working with affine subschemes.
	
	\begin{lemma}\label{chap4:simultaneous-principal-open}
		Let $X$ be a scheme, and let $U, V$ be affine open subschemes of $X$. For each $x \in U \cap V$, there is an open set $W \subseteq U \cap V$ containing $x$ such that $W$ is a principal open subset of $U$ and $V$.
	\end{lemma}
	
	\begin{proof}
		Write $U = \Spec R$ and $V = \Spec S$. There is a principal open subset $D(r) \subseteq \Spec R$ containing $x$ for suitable $r \in R$. Then $D(r) \cap \Spec S$ is open in $\Spec S$, and so there is a principal open subset $D(s) \subseteq D(r) \cap \Spec S$ containing $x$ for suitable $s \in S$. Consider the image of $s$ under the restriction $\varphi\colon\Gamma(\Spec S, \mathcal{O}_X) \to \Gamma(D(r), \mathcal{O}_X) = R_r$. Hence $\varphi(s) = a/r^n$ for suitable $a \in R$ and $n \geq 0$. 
		
		Let $y \in D(r) \cap \Spec S$. Then $y$ corresponds to a prime ideal $\mathfrak{q}_y$ of $S$ and a prime ideal $\mathfrak{p}_yR_r$ of $R_r$, where $\mathfrak{p}_y$ is a prime ideal of $R$ disjoint from the set $\Set{1, r, r^2, \dotsc}$. Using the anti-equivalence of the categories $(\Rings)$ and $(\Aff)$ (\cref{chap3:antiequivalence-of-affine-schemes-and-rings}), they are related by the formula $\varphi^{-1}(\mathfrak{p}_yR_r) = \mathfrak{q}_y$. Therefore
		\begin{equation*}
			s \notin \mathfrak{q}_y \quad\text{ if and only if }\quad \frac{a}{r^n} \notin \mathfrak{p}_yR_r.
		\end{equation*}
		Note that $1/r^n$ is a unit of $R_r$, and so $s \notin \mathfrak{q}_y$ if and only if $a/1 \notin\mathfrak{p}_yR_r$, if and only if $a \notin \mathfrak{p}_y$. Hence we have the equalities
		\begin{equation*}
			D(s) = D(a) \cap D(r) = D(ar).
		\end{equation*}
		Therefore $W \coloneqq D(s)$ is equal to both principal open subsets $\Spec S_s$ and $\Spec R_{ar}$.
	\end{proof}

	\begin{lemma}[Morphism Gluing Lemma]
		Let $X$ and $Y$ be locally ringed spaces, and let $\Set{U_i}_{i \in I}$ be an open covering of $X$. Then a family of morphisms $f_i \colon U_i \to Y$ of locally ringed spaces together form a morphism $f \colon X \to Y$ if and only if the morphisms $f_i \colon U_i \to Y$ and $f_j \colon U_j \to Y$ coincide on $U_i \cap U_j$ for all $i, j \in I$. Furthermore, the resulting morphism $X \to Y$ is unique.
	\end{lemma}
	\begin{proof}
		The only non-trivial part is constructing a morphism $f \colon X \to Y$ from the morphisms $f_i \colon U_i \to Y$. At the level of the underlying topological spaces this is always possible, so we obtain a continuous map $f\colon X \to Y$ such that $f|_{U_i} = f_i$ for all $i \in I$.
		
		We seek to define a morphism of sheaves $f^\flat \colon \mathcal{O}_Y \to f_\ast\mathcal{O}_X$. Let $V \subseteq Y$ be an open subset, so that $f^{-1}(V)$ is an open subset of $X$. For each $i \in I$, we are given a map
		\begin{equation*}
			\mathcal{O}_Y(V) \to (f_i)_{\ast}\mathcal{O}_{X|U_i}(V) = \mathcal{O}_{X}(U_i \cap f^{-1}(V)).
		\end{equation*}
		For each $s \in \mathcal{O}_Y(V)$, let $s_i$ denote the image of $s$ in $\mathcal{O}_{X}(U_i \cap f^{-1}(V))$ under the above map. The idea is to glue the sections $s_i$ together and take the resulting section to be $f^\flat_V(s)$. To make sure that the sections can be glued together, we need to show that
		\begin{equation*}
			s_i|_{U_i \cap U_j \cap f^{-1}(V)} = s_j|_{U_i \cap U_j \cap f^{-1}(V)}
		\end{equation*}
		for all $i, j \in I$. This is implied by our assumption that $f_i$ and $f_j$ coincide on $U_i \cap U_j$. Furthermore, $f^\flat_V(s)$ is well-defined by the uniqueness condition of sheaf axioms.
		
		The only thing left to check is that the maps $ f^\flat_V$ are compatible with restriction maps. Let $W \subseteq V$ be open subsets of $Y$. Then for all $s \in \mathcal{O}_Y(V)$ and $i \in I$,
		\begin{align*}
		f_V^\flat(s)|_{f^{-1}(W) \cap U_i} & = f_V^\flat(s)|_{f^{-1}(V) \cap U_i}|_{f^{-1}(W) \cap U_i} \\
		& = s_i|_{f^{-1}(W) \cap U_i}\\
		& = (s|_W)_i.
		\end{align*}
		By uniqueness, $f^\flat_V(s)|_{f^{-1}(W)} = f^\flat_W(s|_W)$. This proves that $f^\flat$ is compatible with restriction maps and so define a morphism of sheaves $\mathcal{O}_Y \to f_\ast\mathcal{O}_X$. For each $x \in X$, the induced map on stalks $f^\sharp_x$ is plainly a local ring homomorphism because taking stalk is local.
	\end{proof}

	Recall that a ring $R$ is \inlinedef{Noetherian} if it satisfies the \textit{ascending ideal condition}: every chain of ideals
	\begin{equation*}
		I_1 \subseteq I_2 \subseteq \dotsb \subseteq I_n \subseteq \dotsb
	\end{equation*}
	is eventually constant, that is, there is $N \in \NN$ such that $I_N = I_n$ for all $n \geq N$. Equivalently every ideal of $R$ is finitely generated. The corresponding notion for schemes is as follows.
	
	\begin{definition}
		A scheme $X$ is \inlinedef{locally Noetherian} if $X$ admits an open covering $X = \bigcup_i U_i$ by affine open subschemes $U_i$ such that the ring $\mathcal{O}_X(U_i)$ is Noetherian. A scheme is \inlinedef{Noetherian} if it is quasi-compact and locally Noetherian.
	\end{definition}
	
	The following observations are immediate.
	\begin{remark}\label{chap5:remark-noetherian}
		Let $X$ be a locally Noetherian scheme. Suppose that $\Set{U_i}_{i\in I}$ is an open covering of $X$ by affine open subschemes such that $U_i = \Spec A_i$ and $A_i$ is Noetherian for all $i \in I$.
		\begin{enumerate}
			\item Recall that every affine scheme is quasi-compact (\cref{chap2:principal-open-is-quasi-compact}). Hence if $X$ is both affine and locally Noetherian, then it is automatically Noetherian.
			
			\item Localisations of a Noetherian ring are again Noetherian. Consequently, every principal open subset $D(f) = \Spec (A_i)_{f}$ of $U_i$ is Noetherian. Open sets of this form give a basis of $X$, and so $X$ has a basis consisting of Noetherian affine open subschemes.
		\end{enumerate}
	\end{remark}
	
	\begin{lemma}\label{chap5:affine-noetherian}
		An affine scheme $\Spec A$ is Noetherian if and only if the ring $A$ is Noetherian.
	\end{lemma}
	\begin{proof}
		Clearly the condition is sufficient. For the converse, suppose that $\Spec A$ is Noetherian. As previously discussed, we may cover $\Spec A$ by finitely many principal open subsets $D(f_i)$, $i = 1, \dotsc, n$, such that each ring $A_{f_i}$ is Noetherian. Let $I$ be an ideal of $A$. It follows that the ideals $J_i = IA_{f_i}$ of $A_{f_i}$ are finitely generated. The following lemma says that $I$ is finitely generated which concludes the proof.
	\end{proof}
	
	\begin{lemma}
		Let $A$ be a ring and suppose that there are elements $f_1, \dotsc, f_n$ such that $\langle f_1, \dotsc, f_n\rangle = A$. Let $M$ be an $A$-module. Then $M$ is finitely generated if and only if for all $i = 1, \dotsc, n$ the localisation $M_{f_i}$ is a finitely generated $A_{f_i}$-module.
	\end{lemma}
	\begin{proof}
		The forward implication is obvious. To prove the converse, suppose that $M_{f_i}$ is finitely generated as an $A_{f_i}$-module by elements $m_{ij}/(f_i)^{n_{ij}}$, $j = 1, \dotsc, r_i$, $m_{ij} \in M$. Consider the submodule $N \subseteq M$ generated by all $m_{ij}$. Then $N$ is finitely generated, and furthermore, $N_{f_i} = M_{f_i}$ for all $i$. 
		
		We claim that the $A$-module $M/N$ is $0$, so that $M = N$. In order to prove this, let $\mathfrak{p}$ be a prime ideal of $A$. Then one of the $f_i$ is not an element of $\mathfrak{p}$. It then follows that
		\begin{equation*}
			(M/N)_{\mathfrak{p}} = ((M/N)_{f_i})_{\mathfrak{p}} = (M_{f_i}/N_{f_i})_{\mathfrak{p}} = 0.
		\end{equation*}
		Since $(M/N)_{\mathfrak{p}} = 0$ for all prime ideal $\mathfrak{p}$, $M = N$ as claimed.
	\end{proof}
	
	\begin{remark}\label{chap5:scheme-covered-by-Noetherian=affine-is-locally-Noetherian}
		As an immediate consequence of \cref{chap5:affine-noetherian}, we see that if a scheme $X$ can be covered by affine open subschemes $U_i$ such that each $U_i$ is a Noetherian scheme, then $X$ is locally Noetherian.
	\end{remark}
	
	\begin{proposition}
		The following are equivalent for a scheme $X$.
		\begin{enumerate}
			\item $X$ is locally Noetherian.
			\item Each point $x \in X$ has an affine open neighbourhood $U$ such that $\mathcal{O}_X({U})$ is Noetherian.
			\item For every affine open subset $U \subseteq X$ the ring $\mathcal{O}_X(U)$ is Noetherian.
		\end{enumerate}
	\end{proposition}
	\begin{proof}
		Clearly (1) $\Longleftrightarrow$ (2) and (3) $\Rightarrow$ (1) because schemes may be covered by affine open subschemes. In turn we only need to show that (1) implies (3). 
		
		Assume that $X$ is locally Noetherian and $U \subseteq X$ is an affine open subset. By \cref{chap5:remark-noetherian}, $U$ can be written as a union $U = \bigcup_i U_i$ of Noetherian open subschemes which are affine in $X$. For each $i$, \cref{chap4:simultaneous-principal-open} states that the intersection $U \cap U_i$ can be covered by open subsets $W_{ij}$, $j \in J$, such that $W_{ij}$ is principal open in both $U$ and $U_i$. 
		
		Since $U_i$ is Noetherian and $W_{ij}$ is principal open in $U_i$, it follows that $W_{ij}$ is Noetherian (\cref{chap5:remark-noetherian}). Thus $U = \bigcup_{i,j} W_{ij}$ is a union of affine open subschemes of $U$ which are Noetherian, and so $U$ is locally Noetherian by \cref{chap5:scheme-covered-by-Noetherian=affine-is-locally-Noetherian}. Because $U$ is affine, the ring $\mathcal{O}_X(U)$ is Noetherian by \cref{chap5:remark-noetherian} and \cref{chap5:affine-noetherian}.
	\end{proof}
	
	\subsection*{Noetherian Topological Spaces}
	
	There is a topological version of Noetherianness.
	
	\begin{definition}
		A topological space is Noetherian if it satisfies the descending chain condition for closed subsets: every chain of closed subsets
		\begin{equation*}
			Z_1 \supseteq Z_2 \supseteq \dotsb \supseteq Z_n \supseteq \dotsb
		\end{equation*}
		is eventually constant, that is, there is $N \in \NN$ such that $Z_N = Z_n$ for all $n \geq N$.
	\end{definition}
	
	The following proposition can be proved via Zorn's lemma.
	
	\begin{proposition}
		The following are equivalent for a topological space $X$.
		\begin{enumerate}
			\item $X$ is Noetherian.
			\item Every non-empty set of closed subsets of $X$ has a minimal element.
			
			\item Every non-empty set of open subsets of $X$ has a maximal element.
			
			\item Every open subset is quasi-compact.
		\end{enumerate}
	\end{proposition}
	
	\begin{proposition}[Properties of Noetherian spaces]
		Let $X$ be a Noetherian topological space.
		\begin{enumerate}
			\item $X$ has only finitely many irreducible components.
			\item Every subspace of $X$ is Noetherian.
		\end{enumerate}
	\end{proposition}
	
	
	
	\begin{proposition}
		If $X$ is a Noetherian scheme, then $X$ is a Noetherian topological space.
	\end{proposition}
	
	\begin{exercise}
		Prove the propositions.
	\end{exercise}
	
	\begin{remark}
		The converse is not true.
	\end{remark}
	
	
	\section{Generic Points and Irreducible Components}
	
	The generic point of a topological space $X$ is a point $\eta \in X$ such that $\overline{\Set{\eta}} = X$. Just as for affine schemes, there is a one-to-one correspondence between points of a scheme $X$ and its closed irreducible subsets.
	
	\begin{proposition}
		Let $X$ be a scheme. Then there is a bijection
		\begin{align*}
			X & \to \Set*{\text{closed irreducible subsets of $X$}} \\
			x & \mapsto \overline{\Set{x}}.
		\end{align*}
		Hence every closed irreducible subset $Z$ of $X$ contains a unique point $\eta \in Z$ such that $\overline{\Set{\eta}} = Z$.
	\end{proposition}
	
	\begin{proof}
		If $X$ is affine, this is \cref{chap2:correspondence-closed-irreducible-minimal}. For the general case, let $X$ be a scheme and let $Z$ be a closed irreducible subset of $X$. Take an affine open subset $U$ of $X$ such that $Z \cap U \neq \emptyset$. We claim that the closure of $Z \cap U$ in $X$ is equal to $Z$. Indeed, let $K$ denote the closure of $Z \cap U$ in $X$. Then $K \cap Z$ is closed in $Z$, and thus we may express $Z$ as a union of two closed sets
		\begin{equation*}
			Z = (K \cap Z) \cup ((X \setminus U) \cap Z).
		\end{equation*}
		But $Z$ is irreducible, so either $(X \setminus U) \cap Z = Z$ or $K \cap Z = Z$. In the former case, we have $X \setminus U \supseteq Z$, contradicting $Z \cap U \neq \emptyset$. Hence we must have $K \cap Z = Z$, i.e.~$K \subseteq Z$. Since $Z$ is closed, this shows that $K = Z$.
		
		It then follows from \cref{chap2:space-irreducible-iff-closure-irreducible} that $Z \cap U$ is irreducible. Therefore $Z \cap U$ is a closed irreducible subset of an affine scheme $X$. By \cref{chap2:closed-irreducible-subset-of-affine-has-unique-generic-point}, $Z \cap U$ contains a unique generic point $\eta$ such that the closure $\overline{\Set{\eta}}^U$ in $U$ is $Z \cap U$. We claim that the closure of $\Set{\eta}$ in $X$ is $Z$. By topology, we have
		\begin{equation*}
			\overline{\Set{\eta}}^{X} \cap U = \overline{\Set{\eta}}^{U} = Z \cap U.
		\end{equation*}
		Using the fact that $\overline{\Set{\eta}}^{X} \supseteq \overline{\Set{\eta}}^{X} \cap U$, we obtain
		\begin{equation*}
			\overline{\Set{\eta}}^X = \overline{\overline{\Set{\eta}}^{X}}^X \supseteq  \overline{Z \cap U}^X = Z.
		\end{equation*}
		Since $Z$ is a closed set which also contains $\eta$, we conclude that $\overline{\Set{\eta}}^X = Z$.
		
		Now let $z \in Z$ be a generic point. Then $z$ is contained in every affine open subset $U$ which meets $Z$. The uniqueness of generic points in the affine case implies that $z = \eta$.
	\end{proof}
	
	\section{Reduced and Integral Schemes}
	
	\begin{definition}
		A scheme $X$ is \inlinedef{reduced} if the local ring $\mathcal{O}_{X, x}$ is reduced for all $x \in X$.
	\end{definition}
	
	\begin{proposition}\label{chap5:reduced-iff-all-ring-are-reduced}
		A scheme $X$ is reduced if and only if for each $U \subseteq X$ the ring $\mathcal{O}_X(U)$ is reduced.
	\end{proposition}
	\begin{proof}
		First suppose that $X$ is reduced. Let $f \in \mathcal{O}_X(U)$ be such that $f^n = 0$ for some $n \geq 1$. Then for each $x \in U$, $f_x^n = 0$. This implies that $f_x = 0$ because the local ring are reduced. Hence $f = 0$ by the sheaf property.
		
		Conversely, suppose that the ring $\mathcal{O}_X(U)$ are reduced for all open subset $U$. Let $x \in X$ and let $f_x \in \mathcal{O}_{X, x}$ be a germ satisfying $f_x^n = 0$ for some $n \geq 1$. We may find a section $f \in\mathcal{O}_X(U)$ defined on an open neighbourhood $U$ of $x$ lifting $f_x$. Then there is an open neighbourhood $V \subseteq U$ of $x$ such that $f^n|_{V} = 0$. But $\mathcal{O}_X(V)$ is reduced, and thus $f|_{V} = 0$. This implies that $f_x = 0$.
	\end{proof}
	
	\begin{definition}
		A scheme $X$ is \inlinedef{integral} if for every non-empty open subset $U\subseteq X$, the ring $\mathcal{O}_X(U)$ is an integral domain.
	\end{definition}
	
	\begin{proposition}
		A non-empty scheme $X$ is integral if and only if it is reduced and irreducible.
	\end{proposition}
	\begin{proof}
		Suppose $X$ is integral. For each $U\subseteq X$, the ring $\mathcal{O}_X(U)$ is integral, hence reduced. This shows that $X$ is reduced by \cref{chap5:reduced-iff-all-ring-are-reduced}. Now assume that $X$ is reducible. We may find two non-empty disjoint open subsets $U_1, U_2$ of $X$. The sheaf conditions imply that $\mathcal{O}_X(U_1 \cup U_2) = \mathcal{O}_X(U_1) \times \mathcal{O}_X(U_2)$ is a product of two non-trivial rings (\cref{chap4:scheme-empty-section-iff-empty}) which cannot be an integral domain.
		
		For the converse, suppose that the scheme $X$ is reduced and irreducible; \textit{a fortiori} $X$ cannot be empty. We first show that $\mathcal{O}_X(X)$ is an integral domain. Suppose $f, g \in\mathcal{O}_X(X)$ are such that $fg = 0$. For $s \in \mathcal{O}_X(X)$, define
		\begin{equation*}
			V(s) \coloneqq \Set*{x \in X : s(x) = 0} = X \setminus X_s.
		\end{equation*}
		This is a closed subset of $X$ by \cref{chap3:set-where-section-is-non-zero-is-closed}. We also immediately see from the definitions that $V(st) = V(s) \cup V(t)$.
		
		Since $fg = 0$, we have $X = V(fg) = V(f) \cup V(g)$. But $X$ is irreducible, and so without loss of generality we may assume that $V(f) = X$. We claim that $f = 0$. This can be verified locally, therefore we shall consider the restriction of $f$ to affine open subsets of $X$. 
		
		Set $a \coloneqq f|_U$ where $U = \Spec A \subseteq X$. Then $a(x) = f(x) = 0$ for all $x \in U$ is equivalent to $a \in \mathfrak{p}_x$ for all prime ideals $\mathfrak{p}_x$ of $A$. Hence $a \in \bigcap_{\mathfrak{p} \text{ prime}} \mathfrak{p} = \Nil(A)$. By hypothesis the scheme $X$ is reduced, whence $A = \mathcal{O}_X(U)$ is a reduced ring. This implies $\Nil(A) = 0$ forcing $f|_U = a = 0$. Therefore $f = 0$ as desired.
		
		Now let $V$ be any non-empty open subset of $X$ viewed as a scheme $(V, \mathcal{O}_{X|V})$. It is straightforward from the definition that $V$ is reduced as a scheme. Furthermore, $V$ is irreducible because non-empty open subsets of an irreducible topological space are themselves irreducible. The same argument as above reveals that $\mathcal{O}_{X|V}(V) = \mathcal{O}_X(V)$ is an integral domain.
	\end{proof}
	
	\begin{proposition}\label{chap5:integral-scheme-local-ring-is-domain}
		If $X$ is an integral scheme, then the local rings $\mathcal{O}_{X, x}$ are integral domains for all $x \in X$.
	\end{proposition}
	\begin{proof}
		Follows from the fact that non-zero localisations of an integral domain are again integral. 
	\end{proof}
	
	\begin{remark}
		The converse of \cref{chap5:integral-scheme-local-ring-is-domain} is false.
	\end{remark}
	
	\begin{example}[Reduced and Integral Affine Schemes]
		Let $X = \Spec A$ be an affine scheme. Then $X$ is reduced if and only if $A$ is reduced. Similarly, $X$ is integral if and only if $A$ is an integral domain.
	\end{example}
	
	\begin{example}
		Affine algebraic sets are example of reduced schemes.
	\end{example}
	
	\subsection*{Function Field of an Integral Scheme}
	
	In classical algebraic geometry, there is the notion of function field $k(V)$ of an algebraic variety $V \subseteq \AAA^n_k$, where $k$ is algebraically closed. It is defined as the field of fraction of the coordinate ring $k[V]$ which exists because $k[V]$ is an integral domain. Its elements are rational functions defined on an open subset of $V$. The following is an analogue for integral schemes.	
	
	\begin{definition}
		Let $X$ be an integral scheme with the generic point $\eta \in X$. The local ring $\mathcal{O}_{X, \eta}$, which is indeed a field, is called the \inlinedef{function field} of $X$ and is denoted by $K(X)$.
	\end{definition}
	
	\begin{example}
		Let $k$ be a field, so that the scheme $\AAA^1_k = \Spec k[T]$ is an integral scheme. Then the function field $K(\AAA^1_k)$ of $\AAA^1_k$ is by definition $\mathcal{O}_{X, \eta} = k[T]_{(0)} = k(T)$ the field of rational functions.
	\end{example}
	
	The following proposition shows that, when $X$ is an integral scheme, the rings $\mathcal{O}_X(U)$ and the local rings $\mathcal{O}_{X, x}$, $x \in X$, can be viewed as subrings of the function field $K(X)$.
	
	\begin{proposition}
		Let $X$ be an integral scheme with $\eta$ its generic point.
		\begin{enumerate}
			\item If $U = \Spec A$ is a non-empty affine open subscheme of $X$, then $K(X) = \Frac(A)$. If $x \in X$, then $\Frac(\mathcal{O}_{X, x}) = K(X)$.
			
			\item Let $U \subseteq V \subseteq X$ be non-empty open subsets, then the morphisms
			\begin{equation*}
				\Gamma(V, \mathcal{O}_X) \xrightarrow{\res^V_U} \Gamma(U, \mathcal{O}_X) \xrightarrow{f \mapsto f_\eta} K(X)
			\end{equation*}
			are injective.
			
			\item For every non-empty open subset $U \subseteq X$ and for every covering $U = \bigcup_i U_i$ by non-empty open subsets $U_i$, we have
			\begin{equation*}
				\Gamma(U, \mathcal{O}_X) = \bigcap_{i} \Gamma(U_i, \mathcal{O}_X) = \bigcap_{x \in U} \mathcal{O}_{X, x},
			\end{equation*}
			where the intersection takes place in $K(X)$.
		\end{enumerate}
	\end{proposition}
	
	\begin{proof}
		(1.) Since $U$ is non-empty, it contains the generic point $\eta$. By definition, the ring $A$ is an integral domain, and so the zero ideal is the unique generic point of $U$. But the closure of $\eta$ in $U$ is $\overline{\Set{\eta}}^U = U$, and so $\eta$ is also the generic point of $U$. Therefore $\eta$ corresponds to the zero ideal of $A$. This implies that $\Frac(A) = A_{(0)} = \mathcal{O}_{X, \eta} = K(X)$. Furthermore, since $A$ is an integral domain, the fraction field of every local ring $\mathcal{O}_{X, x} = A_{\mathfrak{p}_x}$ is also $K(X)$.
		
		(2.) It suffices to show that $\Gamma(U, \mathcal{O}_X) \xrightarrow{f \mapsto f_\eta} K(X)$ is injective. (Given this, it is an easy exercise to show that $\res^V_U$ is injective for all $U \subseteq V$.) Let $f \in \mathcal{O}_X(U)$ be such that $f_\eta = 0$. Note that we can check that $f = 0$ locally on affine open subschemes of $U$.
		
		Let $W$ be an open non-empty affine subscheme of $W \subseteq U$. Then $W$ contains the generic point $\eta$ and so $0 = f_\eta = (f|_{W})_\eta$. But by (1.) the stalk map $\mathcal{O}_X(W) \to \mathcal{O}_{X, \eta} = K(X)$ is the inclusion $\mathcal{O}_X(W) \to K(W)$ of $\mathcal{O}_X(W)$ into its field of fractions which is injective. Hence $f|_{W} = 0$. Since this holds for every open non-empty affine subscheme of $U$, the sheaf condition implies that $f = 0$.
		
		(3.) First let's prove that $\Gamma(U, \mathcal{O}_X) = \bigcap_{i} \Gamma(U_i, \mathcal{O}_X)$. The inclusion $\subseteq$ follows from the injectivity of inclusion maps proved in (2.). For the inverse inclusion $\supseteq$, note that if $s \in\Gamma(U_i, \mathcal{O}_X)$, then it trivially satisfies the condition to glue to a section in $\Gamma(U, \mathcal{O}_X)$.
		
		Finally we prove that $\Gamma(U, \mathcal{O}_X) = \bigcap_{x \in U} \mathcal{O}_{X, x}$. Let $x \in U$ be contained in an open affine subscheme $W \subseteq U$ such that $W = \Spec A$. Then there are injective ring homomorphisms
		\begin{equation*}
			\Gamma(U, \mathcal{O}_X) \to \Gamma(W, \mathcal{O}_X) = A \to A_{\mathfrak{p}_x} = \mathcal{O}_{X, x}.
		\end{equation*}
		Hence $\Gamma(U, \mathcal{O}_X) \subseteq \mathcal{O}_{X, x}$ for all $x$, and thus the forward inclusion holds. 
		
		For the other inclusion, suppose that $s \in \mathcal{O}_{X, x}$ for all $x \in U$. Let $W \subseteq U$ be a non-empty open affine subscheme and write $W = \Spec A$. Then we have $s \in \Frac(A)$ and $s \in A_{\mathfrak{p}_x}$ for all prime ideal $\mathfrak{p}_x$ of $X$. 
		
		Consider the set $\mathfrak{a} = \Set{h \in A : sh \in A}$. It is easy to check that this is an ideal of $A$. If $\mathfrak{a}$ is proper, then it is contained in a maximal ideal $\mathfrak{m}$. But by assumption $s \in A_{\mathfrak{m}}$ which is equivalent to $s = b/g$ for suitable $b \in A$, $g \notin \mathfrak{m}$. This means that $sg = b \in A$, and so $g \in \mathfrak{a}$, a contradiction. Therefore $\mathfrak{a} = A$, implying that $s = s1 \in A = \Gamma(W, \mathcal{O}_X)$. Since this holds for any non-empty open affine subscheme $W = U$ which form an open cover of $U$, we conclude that $s \in \bigcap \Gamma(W, \mathcal{O}_X) = \Gamma(U, \mathcal{O}_X)$ by the first part.
	\end{proof}

	For schemes over a field $k$, the definition of morphisms locally of finite type simplifies considerably. This is because there is only one possible affine covering of $\Spec k$, namely $\Spec k$ itself. Hence we arrive at the following proposition.

	\begin{definition}
		Let $X$ be a topological space. 
		\begin{enumerate}
			\item A chain 
			\begin{equation*}
				X_0 \supsetneq X_1 \supsetneq \dotsb \supsetneq X_l
			\end{equation*}
			of closed irreducible subsets of $X$ is said to be of \inlinedef{length} $l$.
			
			\item The \inlinedef{dimension} $\dim X$ of $X$ is the supremum of the lengths of all possible of closed irreducible subsets.
		\end{enumerate}
	\end{definition}
	
	A topological space is called \inlinedef{equidimensional} if all of its irreducible components have the same dimension.
	
	\begin{remark}
		The dimension of the empty topological space is set to be $-\infty$. In general the dimensions of non-empty topological spaces can be a non-negative integers or $\infty$. 
	\end{remark}
	
	\begin{lemma}[Properties of Dimension]
		Let $X$ be a topological space.
		\begin{enumerate}
			\item Suppose $Y$ is a subspace of $X$. Then $\dim Y \leq \dim X$.
			
			\item If $X$ is irreducible, $\dim X < \infty$ and $Y \subsetneq X$ is a proper closed subset, then $\dim Y < \dim X$.
			
			\item Let $\Set{U_i}_{i \in I}$ be an open covering of $X$. Then $\dim X = \sup_{i \in I} \dim U_i$.
			
			\item Let $J$ be the set of irreducible components of $X$. Then $\dim X = \sup_{Y \in J} \dim Y$.
		\end{enumerate}
	\end{lemma}
	\begin{proof}
		\begin{enumerate}
			\item Let $Z_0 \supsetneq Z_1 \supsetneq \dotsb \supsetneq Z_l$ be a chain of closed irreducible subsets of $Y$. Taking closure preserves strict inequalities, as one may check. Therefore we obtain the chain $\overline{Z_0} \supsetneq\overline{ Z_1} \supsetneq \dotsb \supsetneq \overline{Z_l}$.
			Since the closure of an irreducible set is irreducible, this proves (1).
			
			\item Under the given assumption, any chains of closed irreducible subset of $Y$ can be extended by adding $X$. This proves (2).
			
			\item By (1) we have $\dim U_i \leq \dim X$, and so $\sup_{i \in I}\dim U_i \leq \dim X$.
			
			\item Obvious.	\qedhere
		\end{enumerate}
	\end{proof}
	
	The \inlinedef{dimension of a scheme} $X$ is simply the dimension of the underlying topological space. For affine schemes this coincides with the Krull dimension. Recall that the Krull dimension of a ring $A$ is defined to be the supremum of the lengths of all chains
	\begin{equation*}
		\mathfrak{p}_0 \subsetneq \mathfrak{p}_1 \subsetneq \dotsb \subsetneq\mathfrak{p}_{l}
	\end{equation*}
	of prime ideals in $A$.
	
	\begin{proposition}
		Let $X = \Spec A$. Then the dimension of $X$ as a scheme is equal to the Krull dimension of $A$.
	\end{proposition}
	\begin{proof}
		Follows from \cref{chap2:correspondence-closed-irreducible-minimal} where it is proved that prime ideals correspond bijectively to irreducible closed subsets of $X$, and that this bijection is order-reversing.
	\end{proof}

	The unique morphism $h$ arising from $u$ and $v$ as in the definition of fibre product will often be denoted by $(u, v)_S$. The notation emphasises the fact that $p \circ (u, v)_S = u$ and $q \circ (u, v)_S = q$, showing that $p$ and $q$ acts like projections.
	
	For familiar categories such as the category of sets and the category of topological spaces, fibre products always exist and can be explictly described.
	
	\begin{example}\label{chap6:fibre-product-in-sets-and-tops}
		Let $f \colon X \to S$ and $g \colon Y \to S$ be functions. Then the set
		\begin{equation*}
			Z = \Set*{(x, y) \in X \times Y : f(x) = g(y)},
		\end{equation*}
		together with projection maps $p \colon (x, y) \mapsto x$ and $q \colon (x, y) \mapsto y$, is the fibre product of $f$ and $g$ in $(\Sets)$.
		
		For the category of topological spaces, equip the set $Z$ above with the subspace topology induced from the product topology of $X \times Y$. The projection maps $p$ and $q$ are then continuous, and together with $Z$ form the fibre product of $f$ and $g$ in $(\Tops)$.
	\end{example}
	
	Keeping the notation of \cref{chap6:fibre-product-in-sets-and-tops}, the fibre product of $f$ and $g$ (in the category of sets) can also be written as a union of products of fibres
	\begin{equation*}
		Z =	X \times_S Y = \bigcup_{s \in S} f^{-1}(s) \times g^{-1}(s),
	\end{equation*}
	hence the name \textit{fibre product}.

	\section{Cartier Divisors on Integral Schemes}
	
	Let $X$ be an integral scheme with the generic point $\eta$ and let $K(X) = \mathcal{O}_{X, \eta}$ be the function field of $X$. Denote by $\mathcal{K}_X$ the constant sheaf with value $K(X)$. Since $X$ is irreducible, $\mathcal{K}_X$ is simply the presheaf $U \mapsto K(X)$ for all non-empty open set $U \subseteq X$, and for $\emptyset \neq V \subseteq U$, the restriction map $\mathcal{K}_X(U) \to \mathcal{K}_X(V)$ is the identity map.
	
	\begin{definition}
		A \inlinedef{Cartier divisor} on an integral scheme $X$ is a tuple $(U_i, f_i)_{i \in I}$ satisfying
		\begin{enumerate}
			\item the sets $U_i$ form an open cover of $X$, and,
			\item $f_i \in K(X)^\times$ are elements such that $f_if_j^{-1} \in \Gamma(U_i \cap U_j, \mathcal{O}_X^\times)$ for all $i, j$.
		\end{enumerate}
		
		Two tuples $(U_i, f_i)_{i \in I}$ and $(V_j, g_j)_{j \in J}$ define the same Cartier divisor if $f_ig_j^{-1} \in \Gamma(U_i \cap V_j, \mathcal{O}_X^{\times})$ for all $i \in I$ and $j \in J$.
	\end{definition}
	
	The set of Cartier divisors on $X$ is denoted by $\Div(X)$. We shall equip a structure of an abelian group on $\Div(X)$ as follows. Given Cartier divisors $(U_i, f_i)_{i \in I}$ and $(V_j, g_j)_{j \in J}$, define their sum to be the Cartier divisor $(U_i \cap V_j, f_i g_j)$. Note that if $i_1, i_2 \in I$ and $j_1, j_2 \in J$, then $f_{i_1} f_{i_2}^{-1} \in \Gamma(U_{i_1} \cap U_{i_2}, \mathcal{O}_X^\times)$ and $g_{j_1} g_{j_2}^{-1} \in \Gamma(V_{j_1} \cap V_{j_2}, \mathcal{O}_X^\times)$. Hence
	\begin{equation*}
		f_{i_1}g_{j_1}(f_{i_2}g_{j_2})^{-1} = f_{i_1}f_{i_2}^{-1}g_{j_1}g_{j_2}^{-1}
	\end{equation*}
	is an element of $\Gamma(U_{i_1} \cap U_{i_2} \cap V_{j_1} \cap V_{j_2}, \mathcal{O}_X^\times)$. Furthermore, this is independent of the choice of the tuples. Indeed, assume that $(U_i, f_i)_{i \in I}$ defines the same divisor as $(U'_k, h_k)_{k \in K}$, and that $(V_j, g_j)_{j \in J}$ and $(V'_m, a_m)_{m \in M}$ define the same Cartier divisor. We then have two tuples
	\begin{equation*}
		(U_i \cap V_j, f_ig_j) \quad\text{and}\quad (U'_k \cap V'_m, h_ka_m).
	\end{equation*}
	It is easily checked that $f_ig_j(h_ka_m)^{-1} \in \Gamma(U_i \cap V_j \cap U'_k \cap V'_m, \mathcal{O}_X^\times)$. This addition is clearly commutative and associative. The identity of $\Div(X)$ is the Cartier divisor $(X, 1)$ and the inverse of the Cartier divisor $(U_i, f_i)_{i \in I}$ is then $(U_i, f_i^{-1})_{i \in I}$. We will use the additive notation for this Abelian group structure on $\Div(X)$.
	
	\begin{remark}
		If $X$ be an integral scheme, then $\Div(X) = \Gamma(X, \mathcal{K}_X^\times/\mathcal{O}_X^\times)$.
	\end{remark}
	
	\begin{definition}
		A Cartier divisor is \inlinedef{principal} if it is equal to the divisor given by $(X, f)$ for some $f \in K(X)^\times$. Two Cartier divisors $D$ and $E$ are \inlinedef{linearly equivalent} if their difference $D - E$ is principal.
	\end{definition}
	
	It is easily seen that the set of principal Cartier divisors is a subgroup of $\Div(X)$ which is isomorphic to $K(X)^\times$. The quotient of $\Div(X)$ by the subgroup of principal divisors is called the \inlinedef{group of divisor classes} on $X$, and is denoted by $\DivCl(X)$.
	
	\begin{proposition}
		Let $X$ be an integral scheme. We have an exact sequence
		\begin{equation*}
			1 \to \Gamma(X, \mathcal{O}_X)^\times \to K(X)^\times \to \Div(X) \to \DivCl(X) \to 0
		\end{equation*}
		of Abelian groups.
	\end{proposition}
	\begin{proof}
		The sequence is exact at $\Gamma(X, \mathcal{O}_X)^\times$ because the map $\Gamma(X, \mathcal{O}_X)^\times \to K(X)^\times$ is injective. Next, if $a \in \Gamma(X, \mathcal{O}_X)^\times$, then the Cartier divisor $(X, a)$ is equal to $(X, 1)$ because $a = a1 \in\Gamma({X, \mathcal{O}_X^\times})$, and so the sequence is exact at $K(X)^\times$. Finally the exactness at $\Div(X)$ and at $\DivCl(X)$ is simply due to the definition of $\DivCl(X)$.
	\end{proof}
	
	\subsection*{Line Bundle attached to a Cartier Divisor}
	
	Let $D = (U_i, f_i)_{i \in I}$ be a Cartier divisor on $X$. We define a line bundle $\mathcal{O}_X(D)$ attached to $D$ by the assignment
	\begin{equation*}
		\Gamma(U, \mathcal{O}_X(D)) = \Set*{f \in K(X) : \text{for all $i$, } f_if \in \Gamma(U_i \cap U, \mathcal{O}_X)}
	\end{equation*}
	for every non-empty open subset $U \subseteq X$. It is easily checked that $\mathcal{O}_X(D)$ is an $\mathcal{O}_X$-module. 
	
	\begin{remark}
		For each $U_i$, consider the restriction $\mathcal{O}_X(D)|_{U_i}$. For each open subset $V \subseteq U_i$, $\Gamma(V, \mathcal{O}_X(D)|_{U_i})$ is the free submodule of $\Gamma(V, \mathcal{O}_X)$ generated by $f_i^{-1}$.
	\end{remark}
	
	Let us temporarily denote this submodule by $M$. Clearly, $M$ is free. For all $rf_i^{-1}$, $r \in \mathcal{O}_X(V)$, and for all $j \in I$, we have
	\begin{equation*}
		f_j(rf_i^{-1}) = rf_jf_i^{-1} \in K(X).
	\end{equation*} 
	By the definition of Cartier divisor, $f_jf_i^{-1} \in \Gamma(U_i \cap U_j, \mathcal{O}_X^\times)$. Since $U_i \cap U_j \cap V = U_j \cap V$, we may view $\Gamma(U_i \cap U_j, \mathcal{O}_X^\times)$ as a subset of $\Gamma(U_j \cap V, \mathcal{O}_X^\times) \subseteq M$, and so we see that $f_j(rf_i^{-1}) \in \Gamma(V, \mathcal{O}_X(D_i)|_{U_i})$. Therefore $M \subseteq \Gamma(V, \mathcal{O}_X(D_i)|_{U_i})$.
	
	Conversely, let $f \in \Gamma(V, \mathcal{O}_X(D_i)|_{U_i})$. Then $f \in K(X)$ and $f_if \in \Gamma(U_i \cap V, \mathcal{O}_X) = \Gamma(V, \mathcal{O}_X)$. Thus $f = (f_if)f_i^{-1} \in M$. This proves the remark.
	
	From the remark we see that $\mathcal{O}_X(D)|_{U_i} \cong \mathcal{O}_{X|U_i}$, and so $\mathcal{O}_X(D)$ is indeed a line bundle on $X$.

	\section{Torsors}
	Let $X$ be a topological space and $G$ a sheaf of groups on $X$. If $T$ is a sheaf of sets on $X$, then we say that $G$ \inlinedef{acts on} $T$ or that $T$ is a \inlinedef{$G$-sheaf} if there is a morphism of sheaves $G \times T \to T$ such that for every open subset $U \subseteq X$, the map $G(U) \times T(U) \to T(U)$ is a left group action of $G(U)$ on $T(U)$.
	
	A \inlinedef{morphism of $G$-sheaves} is a morphism of sheaves $\varphi\colon T \to T'$ such that $\varphi_U \colon T(U) \to T'(U)$ is $G(U)$-equivariant for all open subset $U \subseteq X$. By definition, this means that $\varphi_U(g\cdot t) = g\cdot \varphi_U(t)$ for all open subset $U \subseteq X$, $g \in G(U)$, and $t\in T(U)$.
	
	\begin{definition}
		A $G$-sheaf $T$ is called a \inlinedef{$G$-torsor} if it satisfies the following two properties:
		\begin{enumerate}
			\item The group $G(U)$ acts simply transitively on $T(U)$ for every open subset $U \subseteq X$. (A group $G$ acts \textit{simply transitively} on $X$ if for all $t, s \in X$, there is a unique $g \in G$ such that $g\cdot t = s$.)
			
			\item There exists an open covering $\Set{U_i}_{i \in I}$ of $X$ such that $T(U_i) \neq \emptyset$ for all $i\in I$.
		\end{enumerate}
	\end{definition}
	
	An example of a $G$-torsor is $G$ itself under left multiplication action. It is called the \inlinedef{trivial $G$-torsor}. 
	
	\begin{remark}
		A $G$-torsor $T$ is isomorphic to the trivial $G$-torsor if and only if $T(X) \neq \emptyset$. To see this, take any $t \in T(X)$ and define a morphism $f_U \colon G(U) \to T(U)$ by $g \mapsto gt|_U$ for all open subset $U \subseteq X$ and $g \in G(U)$. The fact that $G(U)$ acts simply transitively implies that each $f_U$ is an isomorphism. The morphisms $f_U$ are compatible with restriction maps and hence together form an isomorphism $f \colon G \to T$ of sheaves.
	\end{remark}
